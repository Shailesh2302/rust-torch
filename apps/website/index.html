Installation

Add to any Rust app (Rust 1.70+):

cargo add rust-torch

Or in Cargo.toml:

[dependencies]
rust-torch = "0.1.0"

Use it:

use rust_torch::{Tensor, nn::{GPT, GPTConfig, Module}};

Monorepo layout: crates/rust-torch (the crate) + apps/website (these docs). Build all with npm run build (turbo).

Quickstart

use rust_torch::Tensor;

let x = Tensor::new(vec![2.0, 3.0], vec![2], true);
let loss = x.mul(&x).sum(); // 13.0
loss.backward();
assert_eq!(x.grad_vec(), vec![4.0, 6.0]);
use rust_torch::nn::{GPT, GPTConfig, Module};
use rust_torch::optim::AdamW;

let cfg = GPTConfig {
    vocab_size: 65, block_size: 16,
    n_layer: 2, n_head: 2, n_embd: 32,
    dropout: 0.0, bias: true,
};
let model = GPT::new(cfg);
let mut opt = AdamW::new(model.parameters(), 3e-3);

let ids = vec![1usize; 2*16];
let tgt = vec![2usize; 2*16];
opt.zero_grad();
let logits = model.forward(&ids, 2, 16); // [2,16,65]
let loss = model.loss(&logits, &tgt);
loss.backward();
opt.step();
// greedy (temp=0) or sampled
let out = model.generate(vec![1, 2, 3], 40, 0.8, Some(20));
// decode with CharTokenizer
use rust_torch::tokenizer::{CharTokenizer, Tokenizer};
let tok = CharTokenizer::build("hello world");
println!("{}", tok.decode(&out));

Examples

Run from the repo root:

cargo run --example sample
cargo run --release --example train_gpt

train_gpt — char-level GPT

use rust_torch::data::TextDataset;
use rust_torch::tokenizer::{CharTokenizer, Tokenizer};

let tok = CharTokenizer::build(&corpus);
let ids = tok.encode(&corpus);
let (train, val) = TextDataset::split_train_val(ids, 16, 0.1);
let (xs, ys, b, t) = train.random_batch(4);
let logits = model.forward(&xs, b, t);
let loss = model.loss(&logits, &ys);

Sampling (temperature / top-k / top-p)

use rust_torch::sampling::sample_next;
let next = sample_next(&last_logits, 0.8, Some(20), None);
